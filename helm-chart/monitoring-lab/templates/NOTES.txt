Access the services:

{{- if eq .Values.grafana.service.type "NodePort" }}

{{- else }}
  Grafana:      kubectl port-forward -n {{ .Values.namespace }} svc/grafana {{ .Values.grafana.service.port }}:{{ .Values.grafana.service.port }}
{{- end }}

{{- if eq .Values.prometheus.service.type "NodePort" }}

{{- else }}

{{- end }}

{{- if eq .Values.alertmanager.service.type "NodePort" }}
  Alertmanager: http://localhost:{{ .Values.alertmanager.service.nodePort }}
{{- else }}

{{- end }}

{{- if eq .Values.jaeger.service.type "NodePort" }}
  Jaeger:       http://localhost:{{ .Values.jaeger.service.nodePort }}
{{- else }}

{{- end }}

{{- if eq .Values.app.service.type "NodePort" }}

{{- else }}

{{- end }}

Check pod status:
  kubectl get pods -n {{ .Values.namespace }}

View logs:
  kubectl logs -n {{ .Values.namespace }} -l app.kubernetes.io/name=app
  kubectl logs -n {{ .Values.namespace }} -l app.kubernetes.io/name=prometheus
  kubectl logs -n {{ .Values.namespace }} -l app.kubernetes.io/name=grafana

To uninstall:
  helm uninstall {{ .Release.Name }} -n {{ .Values.namespace }}
